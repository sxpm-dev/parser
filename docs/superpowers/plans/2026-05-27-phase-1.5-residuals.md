```js
function syntaxErrorAt(input: string, offset: number): SyntaxError {
  let lineStart = offset;
  while (lineStart > 0 && input.charCodeAt(lineStart - 1) !== 0x0a && input.charCodeAt(lineStart - 1) !== 0x0d) lineStart--;
  let lineEnd = offset;
  while (lineEnd < input.length && input.charCodeAt(lineEnd) !== 0x0a && input.charCodeAt(lineEnd) !== 0x0d) lineEnd++;
  let line = 1;
  for (let i = 0; i < lineStart; i++) if (input.charCodeAt(i) === 0x0a) line++;
  const source = input.slice(lineStart, lineEnd);
  const gutter = String(line);
  const column = Math.max(0, offset - lineStart);
  const ch = offset < input.length ? input[offset] : 'EOF';
  const headline = ch === 'EOF'
    ? \`Unexpected end of input at position \${offset}\`
    : \`Unexpected '\${ch}' at position \${offset}\`;
  return new SyntaxError(\`\${headline}
\${gutter} | \${source}
\${' '.repeat(gutter.length)} | \${' '.repeat(column)}^\`);
}
```

Notes:
- The "unexpected byte" is `input[offset]` when within bounds, otherwise the sentinel `EOF`.
- The frame format (line gutter + caret) is unchanged so other tests' frame-match assertions stay green.

## Task 2: UnknownChars enforcement inside `.until()` bodies

**Why:** css test #1 (`parse('foo { color: \`red\` }')`) expects SyntaxError because the CSS schema declares `unknownChars: 'reject'` and backtick (0x60) is not declared in any literal/class/punctuator. v2's `.until(';')` value capture currently scans bytes blindly until the stop, swallowing the backtick.

**Strategy:** Compute, at codegen time, the set of "known leading bytes" — the union of (a) the first byte of every literal/punctuator in the schema, (b) the first byte of every class scanner (ident-start, digit, hex, comment `/`, string `"`/`'`, etc.), and (c) whitespace bytes (0x09, 0x0a, 0x0d, 0x20). When the schema has `unknownChars: 'reject'`, emit a depth-0 byte-class check inside the `.until()` main loop: if the current byte is not a stop, not whitespace, not opening a balanced bracket / string / comment, AND not in the known-leading-bytes set, fail the production (return null, record `state.maxOff`).

- Modify: `packages/compiler/src/emit/v2/span.ts` — `SpanContext` gains `unknownChars` and `knownBytes`; `emitUntilSpan` and `emitRawUntilSpan` emit the guard.
- Modify: `packages/compiler/src/emit/v2/production.ts` — passes new fields into SpanContext (read where SpanContext is constructed).
- Test: `packages/compiler/test/v2-regressions.test.ts`

- [ ] **Step 3: Compute known leading bytes.** Add to `packages/compiler/src/emit/v2/span.ts` (or a small new helper file `packages/compiler/src/emit/v2/known-bytes.ts` if it grows past ~50 lines):

```ts
// Compute the union set of byte values (0..255) that can legitimately start a
// token in this schema: literal/punctuator first byte, class-scanner first
// byte ranges (ident-start, digit, hex, string quotes, comment `/`), plus the
// 4 whitespace bytes. Used by .until() bodies when the schema declares
// unknownChars: 'reject' to fail fast on undeclared bytes.
export function computeKnownLeadingBytes(schema: InferredSchema): Uint8Array {
  const bits = new Uint8Array(256);
  // Whitespace
  bits[0x09] = 1; bits[0x0a] = 1; bits[0x0d] = 1; bits[0x20] = 1;
  // Literals + punctuators
  for (const [text] of schema.literalTypeByText) {
    if (text.length > 0) bits[text.charCodeAt(0) & 0xff] = 1;
  }
  // Classes
  for (const cls of schema.classes) {
    if (cls.kind === 'ident' || cls.kind === 'keyword') {
      // ident-start: A-Z, a-z, _, also $ for js-ident, - for css-ident
      for (let c = 0x41; c <= 0x5a; c++) bits[c] = 1;
      for (let c = 0x61; c <= 0x7a; c++) bits[c] = 1;
      bits[0x5f] = 1;
      if (cls.refinements?.some((r) => r.name === 'css')) bits[0x2d] = 1;
      if (cls.refinements?.some((r) => r.name === 'js')) bits[0x24] = 1;
    } else if (cls.kind === 'number') {
      for (let c = 0x30; c <= 0x39; c++) bits[c] = 1;
      bits[0x2d] = 1; // leading minus
      bits[0x2b] = 1; // leading plus
      bits[0x2e] = 1; // leading dot (decimal)
    } else if (cls.kind === 'hex') {
      bits[0x23] = 1; // '#'
    } else if (cls.kind === 'string') {
      bits[0x22] = 1; bits[0x27] = 1; // " '
    } else if (cls.kind === 'comment') {
      bits[0x2f] = 1; // '/'
    }
  }
  return bits;
}
```

Adjust the class kinds to match the actual `SchemaClass` union in `packages/compiler/src/model.ts`. If a class kind isn't listed here, default to permissive (no extra bytes set) — the guard's job is to catch obviously-undeclared bytes, not to be byte-exhaustive.

## Task 5: Validator-hook codegen — `validate(input, {validate: true})`

**Why:** css tests #4 and #5 expect `validate('color: q w e;', { validate: true })` to throw when per-property validators (`cssColor.validate`, `cssLengthPercentage.validate` declared via `defineValidation`) reject the value. v2's `validate()` is a stub: it parses and reports completion, never invokes the per-field validators.

**Strategy:** Generate a separate validate-time AST walk that runs after parse succeeds. The walk reads `schema.validation.rules` at codegen time and emits a switch on `node.type`, dispatching to validator calls based on `node[rule.matchField]` membership in `rule.matchValues`. Validator throws are caught, wrapped in `SyntaxError` with the source frame from the validated field's offset.

The field's source offset must be tracked. For Declaration nodes, the value is captured in `parse-fast` mode without offsets. Two options:

(a) Re-parse in `raws` mode when `{validate: true}` is passed, walk the raws AST (which carries offsets), validate against the parsed values.

(b) Augment the fast-mode AST to carry minimal offset info — the start offset of every span that has a validator declared against it.

Option (a) is simpler: parse twice in the rare validate case. The cost is negligible since `{validate: true}` is opt-in for tooling/build pipelines, not the hot path. Use it.

```js
function validatorErrorAt(input: string, offset: number, cause: unknown): SyntaxError {
  // Re-use the frame builder from syntaxErrorAt but with a headline that
  // doesn't lead with "Unexpected '<char>'" — validator failures are about
  // value semantics, not byte sequences.
  let lineStart = offset;
  while (lineStart > 0 && input.charCodeAt(lineStart - 1) !== 0x0a && input.charCodeAt(lineStart - 1) !== 0x0d) lineStart--;
  let lineEnd = offset;
  while (lineEnd < input.length && input.charCodeAt(lineEnd) !== 0x0a && input.charCodeAt(lineEnd) !== 0x0d) lineEnd++;
  let line = 1;
  for (let i = 0; i < lineStart; i++) if (input.charCodeAt(i) === 0x0a) line++;
  const source = input.slice(lineStart, lineEnd);
  const gutter = String(line);
  const column = Math.max(0, offset - lineStart);
  const causeMsg = cause instanceof Error ? cause.message : String(cause);
  return new SyntaxError(\`Validation failed at position \${offset}: \${causeMsg}
\${gutter} | \${source}
\${' '.repeat(gutter.length)} | \${' '.repeat(column)}^\`);
}
```

- [ ] **Step 6: Create the validate-emit module.** New file `packages/compiler/src/emit/v2/validate-emit.ts`:

```ts
import type { InferredSchema } from '../../model.ts';
import { jsString, jsIdent } from '../strings.ts';

// Emit the public validate() function and the per-validator import block.
// Returns { imports, body } so the caller can place imports at file head and
// the function body in the public-api section.
export function emitValidate(schema: InferredSchema, rootFn: string): { imports: string; body: string } {
  const rules = schema.validation?.rules ?? [];
  if (rules.length === 0) {
    // No validators declared — fall through to a stub that just parses.
    return {
      imports: '',
      body: emitValidateStub(rootFn),
    };
  }
  // Group rules by nodeType for dispatch efficiency, then by (matchField,
  // validatorModule, validatorExport) so each unique validator imports once.
  const importMap = new Map<string, string>(); // moduleSpec -> local alias
  for (const r of rules) {
    const key = `${r.validator.module}::${r.validator.exportName}`;
    if (!importMap.has(key)) {
      const alias = `__validator_${importMap.size}`;
      importMap.set(key, alias);
    }
  }
  const importLines: string[] = [];
  for (const [key, alias] of importMap) {
    const [mod, name] = key.split('::');
    importLines.push(`import { ${name} as ${alias} } from ${jsString(mod!)};`);
  }
  // Per-nodeType dispatch table.
  const byNode = new Map<string, typeof rules>();
  for (const r of rules) {
    const list = byNode.get(r.nodeType) ?? [];
    list.push(r);
    byNode.set(r.nodeType, list);
  }
  const visitCases: string[] = [];
  for (const [nodeType, nodeRules] of byNode) {
    const literal = nodeTypeLiteral(nodeType);
    const inner: string[] = [];
    for (const r of nodeRules) {
      const valuesArr = '[' + r.matchValues.map((v) => jsString(v)).join(',') + ']';
      const alias = importMap.get(`${r.validator.module}::${r.validator.exportName}`)!;
      inner.push(
        `      if (${valuesArr}.includes(node[${jsString(r.matchField)}])) {`,
        `        try { ${alias}(node[${jsString(r.field)}]); }`,
        `        catch (e) {`,
        `          const off = (node.source && node.source.start && typeof node.source.start.offset === 'number')`,
        `            ? node.source.start.offset : 0;`,
        `          throw validatorErrorAt(input, off, e);`,
        `        }`,
        `      }`,
      );
    }
    visitCases.push(
      `    case ${jsString(literal)}: {`,
      ...inner,
      `      break;`,
      `    }`,
    );
  }
  const body = `export function validate(input: string, options?: ParseOptions): void {
  const fastResult = __parse(input, { mode: 'fast' });
  if (!options?.validate) return;
  const ast = __parse(input, { mode: 'raws' });
  walkAndValidate(input, ast);
}

function walkAndValidate(input: string, node: unknown): void {
  if (node === null || typeof node !== 'object') return;
  if (Array.isArray(node)) {
    for (const child of node) walkAndValidate(input, child);
    return;
  }
  const n = node as Record<string, unknown> & { type?: string };
  switch (n.type) {
${visitCases.join('\n')}
    default: break;
  }
  for (const key in n) {
    const v = (n as Record<string, unknown>)[key];
    if (typeof v === 'object' && v !== null) walkAndValidate(input, v);
  }
}

(validate as unknown as { __parser2Validator?: { module: string; exportName: string } }).__parser2Validator = {
  module: ${jsString(`@parser2/${schema.packageName}`)},
  exportName: 'validate',
};`;
  return { imports: importLines.join('\n'), body };
}

function emitValidateStub(rootFn: string): string {
  return `export function validate(input: string, options?: ParseOptions): void {
  const state: State = { input, len: input.length, pos: 0, maxOff: 0, mode: 'fast', lineStarts: [] };
  const result = ${rootFn}(state);
  if (result === null || state.pos !== input.length) {
    throw syntaxErrorAt(input, state.maxOff > state.pos ? state.maxOff : state.pos);
  }
}`;
}

// Map node-type names from the IR to the runtime `type` field on emitted
// nodes. The compiler lower-cases canonical names ('Declaration' -> 'decl')
// via the same convention used in render emission; reuse it from there if a
// helper exists. For now this hard-codes the css mapping — generalize when
// other schemas adopt validation.
function nodeTypeLiteral(name: string): string {
  if (name === 'Declaration') return 'decl';
  if (name === 'Rule') return 'rule';
  if (name === 'AtRule') return 'atrule';
  if (name === 'Comment') return 'comment';
  return name.toLowerCase();
}
```
